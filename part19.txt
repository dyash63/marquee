  async function fetchCredits(id) {
    const movie = state.movies.find(m => m.id === id);
    if (!movie || !movie.tmdbId || !state.apiKey || movie.cast || movie.castLoading) return;
    movie.castLoading = true; renderList();
    try {
      const path = movie.mediaType === 'series' ? '/tv/' + movie.tmdbId + '/credits' : '/movie/' + movie.tmdbId + '/credits';
      const data = await tmdbGet(path, { language: 'en-US' });
      // Top-billed only — keep it short so the panel doesn't turn into a phone book
      const top = (data.cast || []).slice(0, 6).map(c => c.name).filter(Boolean);
      const m2 = state.movies.find(mm => mm.id === id);
      if (m2) { m2.cast = top; m2.castLoading = false; saveToFirestore(); }
    } catch (e) {
      const m2 = state.movies.find(mm => mm.id === id);
      if (m2) { m2.cast = []; m2.castLoading = false; }
    }
    renderList();
  }

  function toggleDetails(id) {
    if (expandedIds.has(id)) { expandedIds.delete(id); renderList(); return; }
    expandedIds.add(id); renderList();
    const movie = state.movies.find(m => m.id === id);
    if (!movie || !state.apiKey || !movie.tmdbId) return;
    if (movie.mediaType === 'series' && !movie.seasonDetails && !movie.seasonDetailsLoading) fetchSeasonDetails(id);
    if (!movie.cast && !movie.castLoading) fetchCredits(id);
  }

  function applyHashRoute() {
    var h = location.hash || '';
    var m = /^#item-(.+)$/.exec(h);
    if (m) {
      var id = decodeURIComponent(m[1]);
      if (state.movies.some(function (x) { return x.id === id; })) openModal(id);
      else closeModal(true);
    } else if (state.modalId) {
      closeModal(true);
    }
  }


  listEl.addEventListener('click', (e) => {
    const btn = e.target.closest('[data-action]');
    if (!btn) {
      // whole ticket click opens modal (unless clicking interactive child without action)
      const ticket = e.target.closest('.ticket');
      if (ticket && !e.target.closest('button, a')) {
        openModal(ticket.getAttribute('data-ticket-id'));
      }
      return;
    }
    const id = btn.getAttribute('data-id');
    const action = btn.getAttribute('data-action');
    if (action === 'open-modal') { openModal(id); return; }
    if (action === 'toggle-details') { toggleDetails(id); return; }
    if (action === 'refresh-details') { expandedIds.add(id); fetchSeasonDetails(id); return; }
    if (action === 'share') { const movieToShare = state.movies.find(m => m.id === id); if (movieToShare) shareMovie(movieToShare); return; }
    const movie = state.movies.find(m => m.id === id);
    if (!movie) return;
    if (action === 'watched') movie.status = 'watched';
    if (action === 'towatch') movie.status = 'towatch';
    if (action === 'favourite') movie.favourite = !movie.favourite;
    if (action === 'dismiss-update') { movie.hasUpdate = false; movie.updateDetail = ''; }
    if (action === 'remove') {
      state.movies = state.movies.filter(m => m.id !== id);
      if (state.modalId === id) closeModal();
    }
    saveToFirestore(); renderTabs(); renderList(); renderNotifications(); maybeAutoSuggest(); backfillImdbRatings(); backfillGenres();
    if (state.modalId === id) openModal(id); // refresh modal content after status change
  });

  // Modal backdrop / close / actions
  (function setupModalUi() {
    const backdrop = document.getElementById('modal-backdrop');
    const closeBtn = document.getElementById('modal-close');
    if (closeBtn) closeBtn.addEventListener('click', () => closeModal());
    if (backdrop) {
      backdrop.addEventListener('click', (e) => {
        if (e.target === backdrop) closeModal();
      });
      backdrop.addEventListener('click', (e) => {
        const btn = e.target.closest('[data-action]');
        if (!btn || !backdrop.contains(btn)) return;
        const id = btn.getAttribute('data-id');
        const action = btn.getAttribute('data-action');
        const movie = state.movies.find(m => m.id === id);
        if (!movie) return;
        if (action === 'share') { shareMovie(movie); return; }
        if (action === 'watched') movie.status = 'watched';
        if (action === 'towatch') movie.status = 'towatch';
        if (action === 'favourite') movie.favourite = !movie.favourite;
        if (action === 'remove') {
          state.movies = state.movies.filter(m => m.id !== id);
          saveToFirestore(); closeModal(); renderTabs(); renderList(); renderNotifications(); maybeAutoSuggest();
          return;
        }
        saveToFirestore(); renderTabs(); renderList(); renderNotifications(); maybeAutoSuggest();
        openModal(id);
      });
    }
    window.addEventListener('hashchange', applyHashRoute);
    window.addEventListener('popstate', applyHashRoute);
  })();

  // Returns TMDB's raw candidate matches for a query — title, year, media type,
  // and original language for each, so the caller can show them for disambiguation
  // rather than silently guessing at the top hit (which is often wrong for
  // misspellings, or ambiguous for common titles that exist in several
  // languages/years).
  async function fetchSearchCandidatesRaw(query) {
    const multi = await tmdbGet('/search/multi', { query, include_adult: 'false', language: 'en-US', page: 1 });
    return (multi.results || [])
      .filter(r => r.media_type === 'movie' || r.media_type === 'tv' || r.media_type === 'person')
      .map(r => {
        if (r.media_type === 'person') {
          return {
            tmdbId: r.id,
            mediaType: 'person',
            title: r.name || '',
            year: '',
            language: '',
            posterPath: r.profile_path || '',
            genreIds: [],
            knownFor: r.known_for_department || 'Acting'
          };
        }
        return {
          tmdbId: r.id,
          mediaType: r.media_type === 'tv' ? 'series' : 'movie',
          title: r.media_type === 'tv' ? (r.name || '') : (r.title || ''),
          year: yearFromDate(r.media_type === 'tv' ? r.first_air_date : r.release_date),
          language: r.original_language || '',
          posterPath: r.poster_path || '',
          genreIds: r.genre_ids || []
        };
      })
      .filter(c => c.title)
      .slice(0, 10);
  }

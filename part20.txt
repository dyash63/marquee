  function addResultToList(status) {
    const r = state.searchResult;
    const existingIdx = state.movies.findIndex(m => m.title.toLowerCase() === r.title.toLowerCase() && m.year === r.year);
    let finalId, needsBaseline = false;
    if (existingIdx !== -1) {
      const m = state.movies[existingIdx];
      m.status = status; m.platforms = r.platforms || []; m.summary = r.summary || ''; m.posterLink = r.posterLink || '';
      m.mediaType = r.mediaType || 'movie'; m.tmdbId = r.tmdbId || m.tmdbId; m.genres = r.genres || m.genres || [];
      if (r.imdbRating) m.imdbRating = r.imdbRating;
      if (m.favourite === undefined) m.favourite = false;
      finalId = m.id; needsBaseline = m.mediaType === 'series' && !m.latestKnown;
    } else {
      finalId = Date.now() + '-' + Math.random().toString(36).slice(2, 8);
      needsBaseline = (r.mediaType || 'movie') === 'series';
      state.movies.unshift({
        id: finalId, title: r.title, year: r.year || '', summary: r.summary || '', posterLink: r.posterLink || '',
        mediaType: r.mediaType || 'movie', tmdbId: r.tmdbId || null, latestKnown: '', hasUpdate: false,
        platforms: r.platforms || [], status, favourite: false, addedAt: new Date().toISOString(), genres: r.genres || [],
        imdbRating: r.imdbRating || undefined
      });
    }
    saveToFirestore();
    state.searchState = 'idle'; state.searchResult = null; input.value = '';
    state.activeTab = status; renderResult(); renderTabs(); renderList(); maybeAutoSuggest(); backfillImdbRatings(); backfillGenres();
    if (needsBaseline) establishSeriesBaseline(finalId);
  }

 || '').toLowerCase();
    const summary = (m.summary || '').toLowerCase();
    const genres = (m.genres || []).map(g => String(g).toLowerCase()).join(' ');
    let score = 0;

    for (const t of tokens) {

      else if (title.includes(t)) best = 60;
      else if (fuzzyTokenMatch(title, t)) best = 40;
      if (fuzzyTokenMatch(genres, t)) best = Math.max(best, 35);
      if (fuzzyTokenMatch(summary, t)) best = Math.max(best, 20);

    }

  function highlightText(text, tokens) {
    if (!text) return '';
    if (!tokens || !tokens.length) return escapeHtml(text);
    const sorted = tokens.slice().sort((a, b) => b.length - a.length);
    const parts = [];
    const remaining = String(text);
    const lower = remaining.toLowerCase();
    let i = 0;
    while (i < remaining.length) {
      let hit = null;
      let hitLen = 0;
      for (const t of sorted) {
        if (!t) continue;
        if (lower.startsWith(t, i)) {
          hit = remaining.slice(i, i + t.length);
          hitLen = t.length;
          break;
        }
      }
      if (hit) {
        parts.push('<span class="highlight">' + escapeHtml(hit) + '</span>');
        i += hitLen;
      } else {
        let next = remaining.length;
        for (const t of sorted) {
          if (!t) continue;
          const idx = lower.indexOf(t, i + 1);
          if (idx !== -1 && idx < next) next = idx;
        }
        parts.push(escapeHtml(remaining.slice(i, next)));
        i = next;
      }
    }
    return parts.join('');
  }

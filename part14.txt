  let genreBackfillRunning = false;
  async function backfillGenres() {
    if (!state.apiKey || genreBackfillRunning) return;
    // Titles added before genre tags existed (or added via an older build)
    // never got a `genres` array saved on them — fetch it once per title here
    // so everything already in the library catches up, not just new adds.
    const candidates = state.movies.filter(m => m.tmdbId && (!m.genres || !m.genres.length));
    if (!candidates.length) return;
    genreBackfillRunning = true;
    try {
      const chunkSize = 4;
      for (let i = 0; i < candidates.length; i += chunkSize) {
        const chunk = candidates.slice(i, i + chunkSize);
        await Promise.all(chunk.map(async (m) => {
          try {
            const detailPath = m.mediaType === 'series' ? '/tv/' + m.tmdbId : '/movie/' + m.tmdbId;
            const detail = await tmdbGet(detailPath, { language: 'en-US' });
            const genres = (detail.genres || []).map(g => g.name).filter(Boolean);
            if (genres.length) m.genres = genres;
          } catch (e) { /* skip — will retry on a future backfill */ }
        }));
        renderList();
      }
      saveToFirestore();
    } finally { genreBackfillRunning = false; }
  }
  async function attachImdbRatingsToSuggestions() {
    if (!state.omdbKey || !state.suggestions.length) return;
    await Promise.all(state.suggestions.map(async (s) => {
      s.imdbRating = await getImdbRating(s.mediaType, s.tmdbId);
    }));
    renderSuggestPanel();
  }

  // --- "Suggested for you" recommendation engine ---------------------------
  // Built from titles already in the library (both To watch and Watched):
  // for each library title we pull TMDB's own "recommendations" for it, then
  // rank the combined results by how often + how prominently each candidate
  // showed up across those calls, dropping anything already in the library or
  // previously dismissed. Needs at least MIN_LIBRARY_FOR_SUGGESTIONS titles
  // added before it will suggest anything.

  function suggestionKey(mediaType, tmdbId) { return mediaType + ':' + tmdbId; }

  function maybeAutoSuggest() {
    const libraryCount = state.movies.filter(m => m.tmdbId).length;
    if (state.apiKey && libraryCount >= MIN_LIBRARY_FOR_SUGGESTIONS && state.suggestState === 'idle' && !state.suggestions.length) {
      generateSuggestions();
    } else {
      renderSuggestPanel();
    }
  }

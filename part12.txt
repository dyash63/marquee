  async function tmdbGet(path, params) {
    requireApiKey();
    const url = new URL(TMDB_BASE + path);
    url.searchParams.set('api_key', state.apiKey.trim());
    if (params) Object.keys(params).forEach(k => { if (params[k] != null) url.searchParams.set(k, params[k]); });
    const response = await fetch(url.toString());
    if (!response.ok) {
      let detail = '';
      try { detail = (await response.json()).status_message || ''; } catch (e) {}
      const err = new Error('HTTP ' + response.status + (detail ? ': ' + detail : ''));
      err.status = response.status; throw err;
    }
    return response.json();
  }

  function formatDMY(d) {
    if (!d || typeof d !== 'string') return '';
    const parts = d.slice(0, 10).split('-');
    if (parts.length !== 3) return d;
    const [y, m, day] = parts;
    return day + '-' + m + '-' + y;
  }
  function mapProviders(regionData) {
    if (!regionData) return [];
    const out = [], seen = new Set();
    function add(list, type) {
      (list || []).forEach(p => {
        const key = p.provider_name + type;
        if (seen.has(key)) return; seen.add(key);
        out.push({ name: p.provider_name, type });
      });
    }
    add(regionData.flatrate, 'stream'); add(regionData.ads, 'free'); add(regionData.free, 'free');
    add(regionData.rent, 'rent'); add(regionData.buy, 'buy');
    return out;
  }

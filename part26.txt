  // Keyboard shortcuts
  document.addEventListener('keydown', (e) => {
    // don't steal when typing in inputs (except Escape handled by trap)
    const tag = (e.target && e.target.tagName) || '';
    const typing = tag === 'INPUT' || tag === 'TEXTAREA' || tag === 'SELECT' || (e.target && e.target.isContentEditable);
    if (e.key === '/' && !typing && !e.metaKey && !e.ctrlKey && !e.altKey) {
      e.preventDefault();
      const ls = document.getElementById('list-search');
      if (ls && state.activeTab !== 'suggested') { ls.focus(); ls.select(); }
      return;
    }
    if (state.modalId) return; // modal trap handles Escape
    if (typing) return;
    if (state.activeTab === 'suggested') return;
    const tickets = listEl.querySelectorAll('.ticket');
    if (!tickets.length) return;
    if (e.key === 'ArrowDown') {
      e.preventDefault();
      state.kbIndex = Math.min(state.kbIndex + 1, tickets.length - 1);
      if (state.kbIndex < 0) state.kbIndex = 0;
      renderList();
      const el = listEl.querySelector('.ticket.kb-focused');
      if (el) el.scrollIntoView({ block: 'nearest', behavior: 'smooth' });
    } else if (e.key === 'ArrowUp') {
      e.preventDefault();
      state.kbIndex = Math.max(state.kbIndex - 1, 0);
      renderList();
      const el = listEl.querySelector('.ticket.kb-focused');
      if (el) el.scrollIntoView({ block: 'nearest', behavior: 'smooth' });
    } else if (e.key === 'Enter' && state.kbIndex >= 0) {
      e.preventDefault();
      const el = listEl.querySelector('.ticket.kb-focused');
      if (el) openModal(el.getAttribute('data-ticket-id'));
    }
  });

      openSuggestionModal(card.getAttribute('data-suggest-key'));
    }
  });

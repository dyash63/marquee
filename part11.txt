  function userDocRef() { return state.user ? db.collection('users').doc(state.user.uid) : null; }

  async function loadFromFirestore() {
    const ref = userDocRef();
    if (!ref) return;
    try {
      const snap = await ref.get();
      if (snap.exists) {
        const data = snap.data();
        if (Array.isArray(data.movies)) state.movies = data.movies;
        if (data.region) state.region = data.region;
        if (typeof data.apiKey === 'string' && data.apiKey) {
          state.apiKey = data.apiKey;
          state.apiKeyFromCloud = true;
        }
        if (typeof data.omdbKey === 'string' && data.omdbKey) {
          state.omdbKey = data.omdbKey;
          state.omdbKeyFromCloud = true;
        }
      } else {
        const local = loadMoviesLocal();
        if (local.length) { state.movies = local; await saveToFirestore(true); }
      }
    } catch (e) {
      console.error(e);
      state.movies = loadMoviesLocal();
    }
  }

  async function saveToFirestore(immediate) {
    if (!state.user) return;
    const doSave = async () => {
      const ref = userDocRef();
      if (!ref) return;
      try {
        await ref.set({ movies: state.movies, region: state.region, apiKey: state.apiKey || '', omdbKey: state.omdbKey || '', updatedAt: firebase.firestore.FieldValue.serverTimestamp() }, { merge: true });
        trySave('movies', JSON.stringify(state.movies));
        trySave('region', state.region);
        saveApiKey(state.apiKey);
        saveOmdbKey(state.omdbKey);
        clearStorageBanner();
      } catch (e) { console.error(e); showStorageBanner(); }
    };
    if (immediate) { clearTimeout(saveTimer); await doSave(); }
    else { clearTimeout(saveTimer); saveTimer = setTimeout(doSave, 600); }
  }

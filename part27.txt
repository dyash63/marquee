  function loadImageEl(src) {
    return new Promise((resolve, reject) => {
      const img = new Image();
      img.crossOrigin = 'anonymous';
      img.onload = () => resolve(img);
      img.onerror = () => reject(new Error('image load failed'));
      img.src = src;
    });
  }
  function roundRectPath(ctx, x, y, w, h, r) {
    ctx.beginPath();
    ctx.moveTo(x + r, y);
    ctx.arcTo(x + w, y, x + w, y + h, r);
    ctx.arcTo(x + w, y + h, x, y + h, r);
    ctx.arcTo(x, y + h, x, y, r);
    ctx.arcTo(x, y, x + w, y, r);
    ctx.closePath();
  }
  function truncateToWidth(ctx, text, maxWidth) {
    if (ctx.measureText(text).width <= maxWidth) return text;
    let t = text;
    while (t.length > 1 && ctx.measureText(t + '…').width > maxWidth) t = t.slice(0, -1);
    return t + '…';
  }
  // Draws left-to-right pill "chips" (genres, platforms), wrapping to a new row
  // when a chip would overflow maxWidth. Returns the y position after the chips.
  function drawChips(ctx, items, x, y, maxWidth, fontPx, textColor, bgColor) {
    if (!items || !items.length) return y;
    const padX = 10, padY = 5, gap = 8, lineHeight = fontPx + padY * 2 + 6;
    ctx.font = '700 ' + fontPx + 'px Inter, sans-serif';
    ctx.textBaseline = 'alphabetic';
    let curX = x, curY = y;
    items.forEach(text => {
      const w = ctx.measureText(text).width + padX * 2;
      if (curX !== x && curX + w > x + maxWidth) { curX = x; curY += lineHeight; }
      ctx.fillStyle = bgColor;
      roundRectPath(ctx, curX, curY, w, fontPx + padY * 2, (fontPx + padY * 2) / 2);
      ctx.fill();
      ctx.fillStyle = textColor;
      ctx.fillText(text, curX + padX, curY + fontPx + padY - 3);
      curX += w + gap;
    });
    return curY + lineHeight;
  }
